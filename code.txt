# SEEDING
# ============================================================
def set_seed(seed):
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed_all(seed)
        torch.backends.cudnn.deterministic = True
        torch.backends.cudnn.benchmark = False

# BASE ENV FACTORY

def make_base_env(hybrid: bool, env_type: str, gui: bool):
    if hybrid:
        return TB3HybridEnv(env_type=env_type, gui=gui)
    else:
        return TB3Env(env_type=env_type, gui=gui)


# ============================================================
# VEC ENV FACTORY FOR SUBPROC
# ============================================================
def make_env_fn(hybrid, seed, rank, algorithm, env_type, gui):
    def _init():
        env_seed = seed + rank
        random.seed(env_seed)
        np.random.seed(env_seed)
        env = make_base_env(hybrid=hybrid, env_type=env_type, gui=gui)
        env.reset(seed=env_seed)
        if algorithm in ['lstm-sac', 'tx-sac', 'cnn-gru-sac']:
            env = TemporalObservationWrapper(env)
        return env
    return _init

def get_policy_kwargs(algorithm, feature_dim=256):
    policy_kwargs = None
    if algorithm == 'cnn-gru-sac':
        try:
            from policies.cnn_gru_extractor import SB3_CNNGRU_Extractor
            policy_kwargs = dict(
                features_extractor_class=SB3_CNNGRU_Extractor,
                features_extractor_kwargs=dict(features_dim=feature_dim),
                net_arch=dict(pi=[feature_dim], qf=[feature_dim])
            )
            print(f"[Policy] Using CNN-GRU extractor with features_dim={feature_dim}")
        except ImportError:
            print("WARNING: CNN-GRU extractor not found. Falling back to MLP.")
            return None
    elif algorithm == 'lstm-sac':
        try:
            from policies.lstm_extractor import SB3_LSTM_Extractor
            policy_kwargs = dict(
                features_extractor_class=SB3_LSTM_Extractor,
                features_extractor_kwargs=dict(features_dim=feature_dim),
                net_arch=dict(pi=[feature_dim], qf=[feature_dim])
            )
            print(f"[Policy] Using LSTM extractor with features_dim={feature_dim}")
        except ImportError:
            print("WARNING: LSTM extractor not found. Falling back to MLP.")
            return None
    elif algorithm == 'tx-sac':
        try:
            from policies.transformer_extractor import SB3_Transformer_Extractor
            policy_kwargs = dict(
                features_extractor_class=SB3_Transformer_Extractor,
                features_extractor_kwargs=dict(features_dim=feature_dim),
                net_arch=dict(pi=[feature_dim], qf=[feature_dim])
            )
            print(f"[Policy] Using Transformer extractor with features_dim={feature_dim}")
        except ImportError:
            print("WARNING: Transformer extractor not found. Falling back to MLP.")
            return None
    return policy_kwargs

def get_git_hash():
    try:
        result = subprocess.run(['git', 'rev-parse', 'HEAD'],
                                capture_output=True, text=True, cwd=os.path.dirname(__file__))
        if result.returncode == 0:
            return result.stdout.strip()[:8]
    except:
        pass
    return "unknown"

# env/tb3_env.py – COMPLETE CORRECTED VERSION
# Uses the same RewardCalculator as TB3HybridEnv for consistency

import gym
import numpy as np
import pybullet as p
import pybullet_data
import random
import time
from gym import spaces
import config
from config import (
    TIME_STEP, MAX_STEPS, NUM_LIDAR, LIDAR_RANGE, LIDAR_OFFSET,
    GOAL_THRESHOLD, COLLISION_DIST, LINEAR_VEL, ANGULAR_VEL,
    WHEEL_RADIUS, WHEEL_BASE, STATIC_OBSTACLES, DYNAMIC_CONFIG, ENV_TYPE,
    GUI
)

# Import the same RewardCalculator used by TB3HybridEnv
from env.reward import RewardCalculator


class TB3Env(gym.Env):
    """Base TurtleBot3 environment – Gymnasium compatible (5‑value step).
    
    Uses the same RewardCalculator as TB3HybridEnv for consistent reward
    computation across end-to-end and hybrid architectures.
    """

    def __init__(self, env_type=ENV_TYPE, gui=GUI):
        super().__init__()
        self.env_type = env_type
        self.gui = gui
        self.client = p.connect(p.GUI if gui else p.DIRECT)
        p.setAdditionalSearchPath(pybullet_data.getDataPath())

        # Robot handles
        self.robot_id = None
        self.left_wheel = None
        self.right_wheel = None
        self.goal_id = None
        self.static_obstacles = []
        self.dynamic_obstacles = []

        # World bounds
        self.world_bounds = [-3.0, 3.0, -3.0, 3.0]

        # State
        self.step_count = 0
        self.prev_dist_to_goal = None
        self.goal_pos = np.array([1.5, 1.5, 0.0])

        # ========== USE SAME REWARD CALCULATOR AS HYBRID ==========
        self.reward_calc = RewardCalculator()

        # Build world (will be fully reset in reset)
        self._load_world()
        self._load_robot()
        self._place_goal()
        self._spawn_static_obstacles()
        self._init_dynamic_obstacles()

        # Observation space: LiDAR (36) + v,w (2) + dx,dy,dist (3) = 41
        self.obs_dim = NUM_LIDAR + 2 + 3
        self.observation_space = spaces.Box(low=-np.inf, high=np.inf,
                                            shape=(self.obs_dim,), dtype=np.float32)
        self.action_space = spaces.Box(low=-1.0, high=1.0, shape=(2,), dtype=np.float32)

        self.LIDAR_RANGE = LIDAR_RANGE

    # ------------------------------------------------------------------
    # Setup methods
    # ------------------------------------------------------------------
    def _load_world(self):
        p.resetSimulation()
        p.setRealTimeSimulation(0)
        p.setPhysicsEngineParameter(fixedTimeStep=TIME_STEP, numSubSteps=1, numSolverIterations=10)
        p.setGravity(0, 0, -9.8)
        p.loadURDF("plane.urdf")

    def _load_robot(self):
        start_pos = [0.0, 0.0, 0.0]
        start_ori = [0, 0, 0, 1]
        self.robot_id = p.loadURDF(
            "assets/turtlebot3/turtlebot3_description/urdf/turtlebot3_burger.urdf",
            start_pos, start_ori
        )
        for i in range(p.getNumJoints(self.robot_id)):
            name = p.getJointInfo(self.robot_id, i)[1].decode()
            if "wheel_left" in name:
                self.left_wheel = i
            elif "wheel_right" in name:
                self.right_wheel = i
        assert self.left_wheel is not None and self.right_wheel is not None

    def _place_goal(self):
        vis = p.createVisualShape(p.GEOM_SPHERE, radius=0.1, rgbaColor=[0,1,0,1])
        self.goal_id = p.createMultiBody(baseMass=0, baseVisualShapeIndex=vis,
                                         basePosition=self.goal_pos.tolist())

    def _spawn_static_obstacles(self):
        self.static_obstacles = []
        box_collision = p.createCollisionShape(p.GEOM_BOX, halfExtents=[0.2,0.2,0.1])
        box_visual = p.createVisualShape(p.GEOM_BOX, halfExtents=[0.2,0.2,0.1], rgbaColor=[1,0,0,1])
        for x, y in STATIC_OBSTACLES:
            body = p.createMultiBody(baseMass=0,
                                     baseCollisionShapeIndex=box_collision,
                                     baseVisualShapeIndex=box_visual,
                                     basePosition=[x, y, 0.1])
            self.static_obstacles.append(body)

    def _init_dynamic_obstacles(self):
        self.dynamic_obstacles = []
        cfg = DYNAMIC_CONFIG.get(self.env_type, {'num':0})
        for _ in range(cfg.get('num', 0)):
            x = random.uniform(-2.5, 2.5)
            y = random.uniform(-2.5, 2.5)
            col = p.createCollisionShape(p.GEOM_SPHERE, radius=0.1)
            vis = p.createVisualShape(p.GEOM_SPHERE, radius=0.1, rgbaColor=[0,0,1,1])
            body = p.createMultiBody(baseMass=1, baseCollisionShapeIndex=col,
                                     baseVisualShapeIndex=vis, basePosition=[x,y,0.1])
            vel = [random.uniform(-cfg.get('speed',0.2), cfg.get('speed',0.2)),
                   random.uniform(-cfg.get('speed',0.2), cfg.get('speed',0.2)), 0]
            self.dynamic_obstacles.append({'id': body, 'pos': [x,y,0.1], 'vel': vel})

    def _update_dynamic_obstacles(self):
        cfg = DYNAMIC_CONFIG.get(self.env_type, {'num':0})
        bounds = cfg.get('bounds', [-3,3,-3,3])
        for obs in self.dynamic_obstacles:
            obs['pos'][0] += obs['vel'][0] * TIME_STEP
            obs['pos'][1] += obs['vel'][1] * TIME_STEP
            if obs['pos'][0] < bounds[0] or obs['pos'][0] > bounds[1]:
                obs['vel'][0] *= -1
            if obs['pos'][1] < bounds[2] or obs['pos'][1] > bounds[3]:
                obs['vel'][1] *= -1
            p.resetBasePositionAndOrientation(obs['id'], obs['pos'], [0,0,0,1])

    # ------------------------------------------------------------------
    # Core Gym API (Gymnasium style)
    # ------------------------------------------------------------------
    def reset(self, *, seed=None, options=None):
        """Reset the environment. Returns (obs, info)."""
        # Optionally set random seed
        if seed is not None:
            random.seed(seed)
            np.random.seed(seed)

        self.step_count = 0
        self._load_world()
        self._load_robot()
        
        # Random goal (avoid too close to start)
        while True:
            gx = random.uniform(-2.5, 2.5)
            gy = random.uniform(-2.5, 2.5)
            if np.hypot(gx, gy) > 0.8:
                self.goal_pos = np.array([gx, gy, 0.0])
                break
        self._place_goal()
        self._spawn_static_obstacles()
        self._init_dynamic_obstacles()

        # Random start pose
        x = random.uniform(-0.5, 0.5)
        y = random.uniform(-0.5, 0.5)
        yaw = random.uniform(-np.pi, np.pi)
        p.resetBasePositionAndOrientation(self.robot_id, [x,y,0.0],
                                          p.getQuaternionFromEuler([0,0,yaw]))

        # Settling loop
        for _ in range(20):
            p.stepSimulation()
            if self.gui:
                time.sleep(TIME_STEP)

        # Initialize distance tracking
        self.prev_dist_to_goal = self._distance_to_goal()
        
        # Reset reward calculator's internal state for new episode
        self.reward_calc.reset_step_count()
        
        obs = self._get_observation()
        info = {}
        return obs, info

    def step(self, action):

    """
    Simulate 2D LiDAR using ray casting.
    Returns normalized distances in [0,1] (0 = obstacle at robot, 1 = max range).
    Includes optional Gaussian noise and dropout for sim‑to‑real robustness.
    """
    base_pos, base_orn = p.getBasePositionAndOrientation(robot_id)
    yaw = p.getEulerFromQuaternion(base_orn)[2]

    # Angles evenly spaced around 360°, avoiding duplicate at +π
    angles = np.linspace(-np.pi, np.pi, num_rays, endpoint=False)
    height = 0.15  # LiDAR height above ground

    ray_from = []
    ray_to = []

    for angle in angles:
        ray_angle = yaw + angle
        fx = base_pos[0] + lidar_offset * np.cos(ray_angle)
        fy = base_pos[1] + lidar_offset * np.sin(ray_angle)
        fz = base_pos[2] + height

        tx = base_pos[0] + max_distance * np.cos(ray_angle)
        ty = base_pos[1] + max_distance * np.sin(ray_angle)
        tz = fz

        ray_from.append([fx, fy, fz])
        ray_to.append([tx, ty, tz])

    # Batch ray test
    results = p.rayTestBatch(ray_from, ray_to)

    distances = []
    for i, res in enumerate(results):
        hit_fraction = res[2]   # 0 = at start, 1 = at end
        hit_obj = res[0]
        dist = hit_fraction * max_distance
        # Ignore self hits
        if hit_obj == robot_id:
            dist = max_distance
        # Normalize to [0,1]
        norm = dist / max_distance
        distances.append(norm)

        # Optional visualization (debug)
        if visualize and i % ray_skip == 0:
            hit_point = res[3] if hit_fraction < 1.0 else ray_to[i]
            color = [1, 0, 0] if dist < 0.5 else [0, 1, 0]
            p.addUserDebugLine(ray_from[i], hit_point, color, lifeTime=0.05)

import torch.nn as nn
from stable_baselines3.common.torch_layers import BaseFeaturesExtractor
from config import OBSERVATION_HISTORY, NUM_LIDAR

class SB3_CNNGRU_Extractor(BaseFeaturesExtractor):
    """
    Spatio‑temporal feature extractor:
    - 1D CNN per LiDAR scan (spatial)
    - GRU over time (temporal)
    - Concatenates other features (velocities, waypoint) at each timestep
    """
    def __init__(self, observation_space, features_dim=256):
        super().__init__(observation_space, features_dim)

        self.history = OBSERVATION_HISTORY
        # Per‑timestep observation: LiDAR (NUM_LIDAR) + vel (2) + waypoint (3)
        self.lidar_dim = NUM_LIDAR
        self.other_dim = observation_space.shape[0] // self.history - self.lidar_dim
        # Should be 5 (2 velocities + 3 waypoint)

        # Spatial encoder: 1D CNN for LiDAR rays (input channels=1, length=36)
        self.cnn = nn.Sequential(
            nn.Conv1d(1, 16, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.Conv1d(16, 32, kernel_size=3, padding=1),
            nn.ReLU(),
            nn.Flatten()
        )
        cnn_out = 32 * self.lidar_dim   # e.g., 32 * 36 = 1152

        # GRU over time: input size = CNN features + other features
        self.gru = nn.GRU(
            input_size=cnn_out + self.other_dim,
            hidden_size=128,
            batch_first=True
        )

        # Final projection to features_dim
        self.fc = nn.Linear(128, features_dim)

    def forward(self, observations):
        """
        observations: (batch, history * (lidar_dim + other_dim))
        """
        batch = observations.shape[0]
        # Reshape to (batch, history, lidar_dim + other_dim)
        seq = observations.view(batch, self.history, self.lidar_dim + self.other_dim)

        # Split LiDAR and other features
        lidar_seq = seq[:, :, :self.lidar_dim]          # (B, H, 36)
        other_seq = seq[:, :, self.lidar_dim:]          # (B, H, other_dim)

        # Process each timestep with CNN
        # Flatten batch and time: (B*H, 1, 36)
        bh = batch * self.history
        lidar_flat = lidar_seq.reshape(bh, 1, self.lidar_dim)
        cnn_out = self.cnn(lidar_flat)                  # (B*H, cnn_out)
        cnn_out = cnn_out.reshape(batch, self.history, -1)  # (B, H, cnn_out)

        # Concatenate with other features
        combined = torch.cat([cnn_out, other_seq], dim=2)   # (B, H, cnn_out+other_dim)

        # GRU over time
        _, h_n = self.gru(combined)                     # h_n: (1, B, 128)
        features = self.fc(h_n.squeeze(0))              # (B, features_dim)
        return features

import numpy as np
import random

class RRTPlanner:
    def __init__(self, static_obstacles, bounds=[-3, 3, -3, 3], step_size=0.2, max_iter=1000):
        self.obstacles = static_obstacles
        self.bounds = bounds
        self.step_size = step_size
        self.max_iter = max_iter
        self.start = None
        self.goal = None
        self.path = None

    def set_start(self, start_xy):
        self.start = np.array(start_xy[:2])

    def set_goal(self, goal_xy):
        self.goal = np.array(goal_xy[:2])

    def is_collision(self, p):
        for obs in self.obstacles:
            if np.linalg.norm(p - np.array(obs[:2])) < 0.35:
                return True
        if p[0] < self.bounds[0] or p[0] > self.bounds[1] or p[1] < self.bounds[2] or p[1] > self.bounds[3]:
            return True
        return False

    def nearest(self, sample):
        distances = [np.linalg.norm(node - sample) for node in self.nodes]
        return np.argmin(distances)

    def steer(self, from_node, to_node):
        dir_vec = to_node - from_node
        dist = np.linalg.norm(dir_vec)
        if dist < self.step_size:
            return to_node
        return from_node + (dir_vec / dist) * self.step_size

    def plan(self):
        if self.start is None or self.goal is None:
            return None
        self.nodes = [self.start]
        self.parents = [-1]

        for _ in range(self.max_iter):
            if random.random() < 0.05:
                sample = self.goal
            else:
                sample = np.array([
                    random.uniform(self.bounds[0], self.bounds[1]),
                    random.uniform(self.bounds[2], self.bounds[3])
                ])
            nearest_idx = self.nearest(sample)
            new_node = self.steer(self.nodes[nearest_idx], sample)
            if not self.is_collision(new_node):
                self.nodes.append(new_node)
                self.parents.append(nearest_idx)
                if np.linalg.norm(new_node - self.goal) < self.step_size:
                    path = [self.goal]
                    idx = len(self.nodes)-1
                    while idx != -1:
                        path.append(self.nodes[idx])
                        idx = self.parents[idx]
                    self.path = list(reversed(path))
                    return self.path
        return None

    def get_waypoint(self, distance=0.5):
        if self.path is None:
            return self.goal
        cum_dist = 0
        for i in range(len(self.path)-1):
            seg_len = np.linalg.norm(self.path[i+1] - self.path[i])
            if cum_dist + seg_len >= distance:
                ratio = (distance - cum_dist) / seg_len
                waypoint = self.path[i] + ratio * (self.path[i+1] - self.path[i])
                return waypoint
            cum_dist += seg_len
        return self.goal

import numpy as np
import pybullet as p

class DWAPlanner:
    def __init__(self,
                 max_lin=0.22,
                 max_ang=2.84,
                 lin_acc=0.5,
                 ang_acc=3.2,
                 dt=0.1,
                 horizon=1.0,
                 num_samples_v=5,
                 num_samples_w=11,
                 min_clearance=0.15):
        self.max_lin = max_lin
        self.max_ang = max_ang
        self.lin_acc = lin_acc
        self.ang_acc = ang_acc
        self.dt = dt
        self.horizon = horizon
        self.num_samples_v = num_samples_v
        self.num_samples_w = num_samples_w
        self.min_clearance = min_clearance

    def compute_cmd(self, env, goal_xy):
        # Get robot state
        pos = env.get_robot_position()  # returns (x, y, z)
        x, y = pos[0], pos[1]
        # Get orientation
        _, ori = p.getBasePositionAndOrientation(env.robot_id)
        _, _, yaw = p.getEulerFromQuaternion(ori)

        # Current velocities (approximate from env)
        v_curr, w_curr = env.get_robot_velocity()  # returns (linear, angular)

        # Dynamic window
        v_min = max(-self.max_lin, v_curr - self.lin_acc * self.dt)
        v_max = min(self.max_lin,  v_curr + self.lin_acc * self.dt)
        w_min = max(-self.max_ang, w_curr - self.ang_acc * self.dt)
        w_max = min(self.max_ang,  w_curr + self.ang_acc * self.dt)

        vs = np.linspace(v_min, v_max, self.num_samples_v)
        ws = np.linspace(w_min, w_max, self.num_samples_w)

        lidar = env.get_lidar()
        lidar_max = env.LIDAR_RANGE if hasattr(env, "LIDAR_RANGE") else 3.5

        best_score = -1e9
        best_v, best_w = 0.0, 0.0

        for v in vs:
            for w in ws:
                x_sim, y_sim, yaw_sim, clearance = self._simulate(
                    x, y, yaw, v, w, lidar, lidar_max
                )

                if clearance < self.min_clearance:
                    continue

                dist_goal = np.linalg.norm(np.array(goal_xy) - np.array([x_sim, y_sim]))
                heading_score = -dist_goal
                clearance_score = clearance
                speed_score = v

                score = 2.0 * heading_score + 0.5 * clearance_score + 0.5 * speed_score

                if score > best_score:
                    best_score = score
                    best_v, best_w = v, w

        lin_norm = np.clip(best_v / self.max_lin, -1, 1) if self.max_lin > 0 else 0.0
        ang_norm = np.clip(best_w / self.max_ang, -1, 1) if self.max_ang > 0 else 0.0
        return np.array([lin_norm, ang_norm])

    def _simulate(self, x, y, yaw, v, w, lidar_ranges, lidar_max_range):
        t = 0.0
        min_clearance = lidar_max_range
        while t < self.horizon:
            x += v * np.cos(yaw) * self.dt
            y += v * np.sin(yaw) * self.dt
            yaw += w * self.dt
            t += self.dt
            min_clearance = min(min_clearance, np.min(lidar_ranges) * lidar_max_range)
        return x, y, yaw, min_clearance
