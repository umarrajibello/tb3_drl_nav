# ------------------------------------------------------------
# VecEnv factory for SubprocVecEnv
# ------------------------------------------------------------
def make_env_fn(hybrid, seed, rank, algorithm, env_type, gui):
    def _init():
        env_seed = seed + rank
        random.seed(env_seed)
        np.random.seed(env_seed)
        env = make_base_env(hybrid=hybrid, env_type=env_type, gui=gui)
        env.reset(seed=env_seed)
        if algorithm in ['lstm-sac', 'tx-sac', 'cnn-gru-sac']:
            env = TemporalObservationWrapper(env)
        return env
    return _init

# Fully Corrected eval.py
# Works with both old Gym (4‑value step) and Gymnasium (5‑value step)
# Compatible with end‑to‑end and hybrid environments, with or without temporal wrappers.

import argparse
import os
import csv
import time
import random
import numpy as np
import torch
import gym

from stable_baselines3 import SAC, PPO
from stable_baselines3.common.vec_env import DummyVecEnv, VecNormalize

from env.tb3_env import TB3Env
from env.tb3_env_hybrid import TB3HybridEnv
from env.observation_wrapper import TemporalObservationWrapper

from config import (
    OBSERVATION_HISTORY, GOAL_THRESHOLD, COLLISION_DIST,
    LIDAR_RANGE, ANGULAR_VEL
)

# ------------------------------------------------------------
# Seeding for reproducibility
# ------------------------------------------------------------
def set_seed(seed):
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)

# ------------------------------------------------------------
# Base environment factory
# ------------------------------------------------------------
def make_base_env(hybrid: bool, env_type: str, gui: bool):
    """Create the appropriate base environment (TB3Env or TB3HybridEnv)."""
    if hybrid:
        return TB3HybridEnv(env_type=env_type, gui=gui)
    else:
        return TB3Env(env_type=env_type, gui=gui)

# ------------------------------------------------------------
# Unwrap VecNormalize → DummyVecEnv → underlying base env
# ------------------------------------------------------------
def unwrap_original_env(env):
    """
    Recursively unwrap VecNormalize and DummyVecEnv to access the original
    TB3Env or TB3HybridEnv instance. Needed to call get_robot_pose(), get_lidar(), etc.
    """
    if isinstance(env, VecNormalize):
        env = env.venv
    if isinstance(env, DummyVecEnv):
        env = env.envs[0]
    while hasattr(env, "env"):
        env = env.env
    return env

# ------------------------------------------------------------
# Argument parsing
# ------------------------------------------------------------
def parse_args():
    parser = argparse.ArgumentParser()
    parser.add_argument('--model_path', type=str, required=True,
                        help="Path to the trained model .zip file")
    parser.add_argument('--env_type', type=str, default='mixed',
                        choices=['static', 'mixed', 'dense'])
    parser.add_argument('--algorithm', type=str, required=True,
                        choices=['sac', 'ppo', 'cnn-gru-sac', 'lstm-sac', 'tx-sac'])
    parser.add_argument('--hybrid', action='store_true',
                        help="Use hybrid environment (RRT + waypoint)")
    parser.add_argument('--num_episodes', type=int, default=50)
    parser.add_argument('--gui', action='store_true',
                        help="Show PyBullet GUI")
    parser.add_argument('--output', type=str, default='eval_results.csv')
    parser.add_argument('--seed', type=int, default=42)
    return parser.parse_args()

# ------------------------------------------------------------
# Main evaluation
# ------------------------------------------------------------
def main():
    args = parse_args()
    set_seed(args.seed)

    # Update global config (used by environments)
    import config
    config.ENV_TYPE = args.env_type
    config.GUI = args.gui

    # ---------------- CREATE BASE ENVIRONMENT ----------------
    base_env = make_base_env(args.hybrid, args.env_type, args.gui)
    if args.algorithm in ['cnn-gru-sac', 'lstm-sac', 'tx-sac']:
        # Wrap with TemporalObservationWrapper to stack history
        base_env = TemporalObservationWrapper(base_env)
    env = DummyVecEnv([lambda: base_env])

    # ---------------- LOAD NORMALIZATION STATS ----------------
    norm_path = os.path.join(os.path.dirname(args.model_path), "vecnormalize.pkl")
    if os.path.exists(norm_path):
        try:
            env = VecNormalize.load(norm_path, env)
            env.training = False      # disable updates
            env.norm_reward = False   # we want raw rewards for metrics
            print(f"Loaded VecNormalize stats from {norm_path}")
        except Exception as e:
            print(f"Warning: VecNormalize load failed ({e}). Running without normalization.")
    else:
        print("Warning: No VecNormalize stats found.")

    # ---------------- LOAD TRAINED MODEL ----------------
    if args.algorithm in ['sac', 'cnn-gru-sac', 'lstm-sac', 'tx-sac']:
        model = SAC.load(args.model_path, env=env)
    else:  # PPO
        model = PPO.load(args.model_path, env=env)

    metrics = []

    # ---------------- EVALUATION LOOP ----------------
    for ep in range(args.num_episodes):
        obs = env.reset()
        original_env = unwrap_original_env(env)

        # Record start and goal positions
        start_pos = np.array(original_env.get_robot_pose()[:2])
        goal_pos = np.array(original_env.goal_pos[:2])
        straight_line_dist = np.linalg.norm(goal_pos - start_pos)

        done = False
        steps = 0
        path_length = 0.0
        prev_pos = start_pos.copy()
        collision = False
        dynamic_collision = False
        success = False
        angular_velocities = []
        min_distances = []
        stuck_counter = 0
        stuck_events = 0
        start_time = time.time()

        while not done:
            # Get action from policy (deterministic for evaluation)
            action, _ = model.predict(obs, deterministic=True)

            # Record angular velocity for smoothness metric
            ang_vel = action[0][1] * ANGULAR_VEL
            angular_velocities.append(ang_vel)

            # ---------- CRITICAL FIX: handle both 4-value and 5-value step ----------
            step_result = env.step(action)
            if len(step_result) == 5:          # Gymnasium style
                obs, reward, terminated, truncated, info = step_result
                done = terminated or truncated
            else:                              # Old Gym style (4 values)
                obs, reward, done, info = step_result
            # ------------------------------------------------------------------------

            steps += 1

            # Get current robot position (x, y)
            curr_pos = np.array(original_env.get_robot_pose()[:2])

            # Path length (Euclidean distance travelled)
            step_dist = np.linalg.norm(curr_pos - prev_pos)
            path_length += step_dist
            prev_pos = curr_pos

            # Stuck detection: if robot moves less than 1 cm for 50 steps
            if step_dist < 0.01:
                stuck_counter += 1
            else:
                stuck_counter = 0
            if stuck_counter >= 50:
                stuck_events += 1
                stuck_counter = 0

            # LiDAR clearance (minimum distance to any obstacle)
            raw_lidar = np.array(original_env.get_lidar(), dtype=np.float32)
            raw_lidar = np.clip(raw_lidar, 0.0, 1.0)
            min_dist = np.min(raw_lidar) * LIDAR_RANGE
            min_distances.append(min_dist)

   
            # Collision detection – info is a list of dicts (VecEnv)
                       
            if isinstance(info, list) and len(info) > 0:
                # For DummyVecEnv, info[0] contains the dict for the single environment
                env_info = info[0]
                if env_info.get('collision', False):
                    collision = True
                    # Also check dynamic collision if needed
                    try:
                        dyn_dists = original_env.get_dynamic_obstacle_distances()
                        if dyn_dists and min(dyn_dists) < COLLISION_DIST:
                            dynamic_collision = True
                    except AttributeError:
                        pass

        # Episode finished – compute metrics
        time_to_goal = time.time() - start_time
        jerk = np.mean(np.abs(np.diff(angular_velocities))) if len(angular_velocities) > 1 else 0.0
        min_clearance = np.min(min_distances) if min_distances else 0.0
        path_efficiency = straight_line_dist / path_length if path_length > 0 else 0.0

        # Success if reached goal without collision
        if not collision:
            dist_to_goal = np.linalg.norm(curr_pos - goal_pos)
            if dist_to_goal < GOAL_THRESHOLD:
                success = True

        # Store episode results
        metrics.append({
            'episode': ep,
            'steps': steps,
            'time_to_goal': time_to_goal,
            'path_length': path_length,
            'path_efficiency': path_efficiency,
            'smoothness_jerk': jerk,
            'min_clearance': min_clearance,
            'collision': collision,
            'dynamic_collision': dynamic_collision,
            'stuck_events': stuck_events,
            'success': success
        })

        print(f"Episode {ep}: success={success}, steps={steps}, "
              f"time={time_to_goal:.2f}s, eff={path_efficiency:.2f}, "
              f"jerk={jerk:.3f}, clearance={min_clearance:.2f}")

    # ---------------- SAVE RESULTS TO CSV ----------------
    with open(args.output, 'w', newline='') as f:
        writer = csv.DictWriter(f, fieldnames=metrics[0].keys())
        writer.writeheader()
        writer.writerows(metrics)

    # ---------------- PRINT SUMMARY ----------------
    success_rate = np.mean([m['success'] for m in metrics]) * 100
    collision_rate = np.mean([m['collision'] for m in metrics]) * 100
    dynamic_collision_rate = np.mean([m['dynamic_collision'] for m in metrics]) * 100
    avg_steps = np.mean([m['steps'] for m in metrics])
    avg_eff = np.mean([m['path_efficiency'] for m in metrics])

    print("\n===== Evaluation Summary =====")
    print(f"Success rate: {success_rate:.1f}%")
    print(f"Collision rate: {collision_rate:.1f}% (dynamic: {dynamic_collision_rate:.1f}%)")
    print(f"Average steps: {avg_steps:.1f}")
    print(f"Path efficiency: {avg_eff:.2f}")
    print(f"Results saved to {args.output}")

if __name__ == "__main__":
    main()

import gym
import numpy as np
import pybullet as p
import pybullet_data
import random
import time
from gym import spaces

from config import (
    TIME_STEP, MAX_STEPS, NUM_LIDAR, LIDAR_RANGE, LIDAR_OFFSET,
    GOAL_THRESHOLD, COLLISION_DIST, LINEAR_VEL, ANGULAR_VEL,
    WHEEL_RADIUS, WHEEL_BASE, STATIC_OBSTACLES, DYNAMIC_CONFIG, ENV_TYPE,
    GUI
)

class TB3Env(gym.Env):
    """Base TurtleBot3 environment – Gymnasium compatible (5‑value step)."""

    def __init__(self, env_type=ENV_TYPE, gui=GUI):
        super().__init__()
        self.env_type = env_type
        self.gui = gui
        self.client = p.connect(p.GUI if gui else p.DIRECT)
        p.setAdditionalSearchPath(pybullet_data.getDataPath())

        # Robot handles
        self.robot_id = None
        self.left_wheel = None
        self.right_wheel = None
        self.goal_id = None
        self.static_obstacles = []
        self.dynamic_obstacles = []

        # World bounds
        self.world_bounds = [-3.0, 3.0, -3.0, 3.0]

        # State
        self.step_count = 0
        self.prev_dist_to_goal = None
        self.goal_pos = np.array([1.5, 1.5, 0.0])

        # Build world (will be fully reset in reset)
        self._load_world()
        self._load_robot()
        self._place_goal()
        self._spawn_static_obstacles()
        self._init_dynamic_obstacles()

        # Observation space: LiDAR (36) + v,w (2) + dx,dy,dist (3) = 41
        self.obs_dim = NUM_LIDAR + 2 + 3
        self.observation_space = spaces.Box(low=-np.inf, high=np.inf,
                                            shape=(self.obs_dim,), dtype=np.float32)
        self.action_space = spaces.Box(low=-1.0, high=1.0, shape=(2,), dtype=np.float32)

        self.LIDAR_RANGE = LIDAR_RANGE

    # ------------------------------------------------------------------
    # Setup methods (unchanged, same as before)
    # ------------------------------------------------------------------
    def _load_world(self):
        p.resetSimulation()
        p.setRealTimeSimulation(0)
        p.setPhysicsEngineParameter(fixedTimeStep=TIME_STEP, numSubSteps=1, numSolverIterations=10)
        p.setGravity(0, 0, -9.8)
        p.loadURDF("plane.urdf")

    def _load_robot(self):
        start_pos = [0.0, 0.0, 0.0]
        start_ori = [0, 0, 0, 1]
        self.robot_id = p.loadURDF(
            "assets/turtlebot3/turtlebot3_description/urdf/turtlebot3_burger.urdf",
            start_pos, start_ori
        )
        for i in range(p.getNumJoints(self.robot_id)):
            name = p.getJointInfo(self.robot_id, i)[1].decode()
            if "wheel_left" in name:
                self.left_wheel = i
            elif "wheel_right" in name:
                self.right_wheel = i
        assert self.left_wheel is not None and self.right_wheel is not None

    def _place_goal(self):
        vis = p.createVisualShape(p.GEOM_SPHERE, radius=0.1, rgbaColor=[0,1,0,1])
        self.goal_id = p.createMultiBody(baseMass=0, baseVisualShapeIndex=vis,
                                         basePosition=self.goal_pos.tolist())

    def _spawn_static_obstacles(self):
        self.static_obstacles = []
        box_collision = p.createCollisionShape(p.GEOM_BOX, halfExtents=[0.2,0.2,0.1])
        box_visual = p.createVisualShape(p.GEOM_BOX, halfExtents=[0.2,0.2,0.1], rgbaColor=[1,0,0,1])
        for x, y in STATIC_OBSTACLES:
            body = p.createMultiBody(baseMass=0,
                                     baseCollisionShapeIndex=box_collision,
                                     baseVisualShapeIndex=box_visual,
                                     basePosition=[x, y, 0.1])
            self.static_obstacles.append(body)

    def _init_dynamic_obstacles(self):
        self.dynamic_obstacles = []
        cfg = DYNAMIC_CONFIG.get(self.env_type, {'num':0})
        for _ in range(cfg.get('num', 0)):
            x = random.uniform(-2.5, 2.5)
            y = random.uniform(-2.5, 2.5)
            col = p.createCollisionShape(p.GEOM_SPHERE, radius=0.1)
            vis = p.createVisualShape(p.GEOM_SPHERE, radius=0.1, rgbaColor=[0,0,1,1])
            body = p.createMultiBody(baseMass=1, baseCollisionShapeIndex=col,
                                     baseVisualShapeIndex=vis, basePosition=[x,y,0.1])
            vel = [random.uniform(-cfg.get('speed',0.2), cfg.get('speed',0.2)),
                   random.uniform(-cfg.get('speed',0.2), cfg.get('speed',0.2)), 0]
            self.dynamic_obstacles.append({'id': body, 'pos': [x,y,0.1], 'vel': vel})

    def _update_dynamic_obstacles(self):
        cfg = DYNAMIC_CONFIG.get(self.env_type, {'num':0})
        bounds = cfg.get('bounds', [-3,3,-3,3])
        for obs in self.dynamic_obstacles:
            obs['pos'][0] += obs['vel'][0] * TIME_STEP
            obs['pos'][1] += obs['vel'][1] * TIME_STEP
            if obs['pos'][0] < bounds[0] or obs['pos'][0] > bounds[1]:
                obs['vel'][0] *= -1
            if obs['pos'][1] < bounds[2] or obs['pos'][1] > bounds[3]:
                obs['vel'][1] *= -1
            p.resetBasePositionAndOrientation(obs['id'], obs['pos'], [0,0,0,1])

    # ------------------------------------------------------------------
    # Core Gym API (Gymnasium style)
    # ------------------------------------------------------------------
    def reset(self, *, seed=None, options=None):
        """Reset the environment. Returns (obs, info)."""
        # Optionally set random seed
        if seed is not None:
            random.seed(seed)
            np.random.seed(seed)

        self.step_count = 0
        self._load_world()
        self._load_robot()
        # Random goal (avoid too close to start)
        while True:
            gx = random.uniform(-2.5, 2.5)
            gy = random.uniform(-2.5, 2.5)
            if np.hypot(gx, gy) > 0.8:
                self.goal_pos = np.array([gx, gy, 0.0])
                break
        self._place_goal()
        self._spawn_static_obstacles()
        self._init_dynamic_obstacles()

        # Random start pose
        x = random.uniform(-0.5, 0.5)
        y = random.uniform(-0.5, 0.5)
        yaw = random.uniform(-np.pi, np.pi)
        p.resetBasePositionAndOrientation(self.robot_id, [x,y,0.0],
                                          p.getQuaternionFromEuler([0,0,yaw]))

        # Settling loop
        for _ in range(240):
            p.stepSimulation()
            if self.gui:
                time.sleep(TIME_STEP)

        self.prev_dist_to_goal = self._distance_to_goal()
        obs = self._get_observation()
        info = {}   # no additional info
        return obs, info

    def step(self, action):
        """Take an action. Returns (obs, reward, terminated, truncated, info)."""
        self.step_count += 1
        action = np.clip(action, -1.0, 1.0)

        # Wheel velocity control
        linear = action[0] * LINEAR_VEL
        angular = action[1] * ANGULAR_VEL
        left = (linear - (WHEEL_BASE/2.0)*angular) / WHEEL_RADIUS
        right = (linear + (WHEEL_BASE/2.0)*angular) / WHEEL_RADIUS
        p.setJointMotorControl2(self.robot_id, self.left_wheel, p.VELOCITY_CONTROL,
                                targetVelocity=left, force=2.0)
        p.setJointMotorControl2(self.robot_id, self.right_wheel, p.VELOCITY_CONTROL,
                                targetVelocity=right, force=2.0)

        # Physics step (8 sub‑steps)
        for _ in range(8):
            p.stepSimulation()

class RRTPlanner:
    def __init__(self, static_obstacles, bounds=[-3, 3, -3, 3], step_size=0.2, max_iter=1000):
        self.obstacles = static_obstacles
        self.bounds = bounds
        self.step_size = step_size
        self.max_iter = max_iter
        self.start = None
        self.goal = None
        self.path = None

    def set_start(self, start_xy):
        self.start = np.array(start_xy[:2])

    def set_goal(self, goal_xy):
        self.goal = np.array(goal_xy[:2])

    def is_collision(self, p):
        for obs in self.obstacles:
            if np.linalg.norm(p - np.array(obs[:2])) < 0.35:
                return True
        if p[0] < self.bounds[0] or p[0] > self.bounds[1] or p[1] < self.bounds[2] or p[1] > self.bounds[3]:
            return True
        return False

    def nearest(self, sample):
        distances = [np.linalg.norm(node - sample) for node in self.nodes]
        return np.argmin(distances)

    def steer(self, from_node, to_node):
        dir_vec = to_node - from_node
        dist = np.linalg.norm(dir_vec)
        if dist < self.step_size:
            return to_node
        return from_node + (dir_vec / dist) * self.step_size

    def plan(self):
        if self.start is None or self.goal is None:
            return None
        self.nodes = [self.start]
        self.parents = [-1]

        for _ in range(self.max_iter):
            if random.random() < 0.05:
                sample = self.goal
            else:
                sample = np.array([
                    random.uniform(self.bounds[0], self.bounds[1]),
                    random.uniform(self.bounds[2], self.bounds[3])
                ])
            nearest_idx = self.nearest(sample)
            new_node = self.steer(self.nodes[nearest_idx], sample)

                self.nodes.append(new_node)
                self.parents.append(nearest_idx)

                    while idx != -1:
                        path.append(self.nodes[idx])
                        idx = self.parents[idx]

                    return self.path
        return None

class PurePursuitController:
    def __init__(self, env, lookahead=0.5):
        self.env = env
        self.lookahead = lookahead

    def compute_cmd(self, waypoint):
        """Return normalized action [linear, angular] in [-1,1]."""
        robot_pos = self.env.get_robot_position()
        robot_xy = np.array(robot_pos[:2])
        _, ori = p.getBasePositionAndOrientation(self.env.robot_id)
        _, _, robot_yaw = p.getEulerFromQuaternion(ori)

        dx = waypoint[0] - robot_xy[0]
        dy = waypoint[1] - robot_xy[1]
        angle_to_wp = np.arctan2(dy, dx)
        angle_error = angle_to_wp - robot_yaw
        angle_error = np.arctan2(np.sin(angle_error), np.cos(angle_error))

        # Angular control
        k_ang = 2.0
        angular = np.clip(k_ang * angle_error, -ANGULAR_VEL, ANGULAR_VEL)

        # Linear speed based on heading alignment
        heading_factor = max(0.0, 1.0 - abs(angle_error) / np.pi)
        linear = LINEAR_VEL * (0.2 + 0.8 * heading_factor)

        # Normalize to action space [-1,1]
        return np.array([linear / LINEAR_VEL, angular / ANGULAR_VEL])

def run_rrt_purepursuit(env_type, num_episodes=50, gui=False, output_csv="rrt_purepursuit_results.csv"):
